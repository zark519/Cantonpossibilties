function recalculateAllActualRanks() { let weekKeys = new Set(); songs.forEach(s => { if (!s.entryDate) return; const parts = s.entryDate.split('-'); let baseDate = new Date(parts[0], parts[1] - 1, parts[2]); // 1. 重新推算正確的進榜起始週 let startWeekInfo = getWeekDetails(baseDate); s.startWeekKey = startWeekInfo.weekKey; s.startWeekLabel = startWeekInfo.weekLabel; // 2. 強制重刷新每一週的 weekKey 與 weekLabel let currentSun = new Date(startWeekInfo.startDate); s.weeklyData.forEach(w => { let weekInfo = getWeekDetails(currentSun); w.weekKey = weekInfo.weekKey; w.weekLabel = weekInfo.weekLabel; w.year = weekInfo.year; w.weekNum = weekInfo.weekNum; weekKeys.add(w.weekKey); currentSun.setDate(currentSun.getDate() + 7); // 推進 7 天至下一週 }); }); // 3. 重新計算每週的實際排名 (Actual Rank) weekKeys.forEach(wKey => { let weekEntries = []; songs.forEach(s => { let w = s.weeklyData.find(item => item.weekKey === wKey); if (w) weekEntries.push({ song: s, weekData: w, points: w.points, entryDate: new Date(s.entryDate) }); }); weekEntries.sort((a, b) => b.points !== a.points ? b.points - a.points : b.entryDate - a.entryDate); weekEntries.forEach((entry, idx) => entry.weekData.actualRank = idx + 1); }); }
